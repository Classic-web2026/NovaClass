const SHEET_NAME = 'Users'; const HEADERS = ['email', 'password', 'name', 'createdAt']; function doGet(e) { const p = (e && e.parameter) || {}; const cb = p.callback; let result; try { const a = p.action || ''; if (a === 'register') result = register(p); else if (a === 'login') result = login(p); else result = { status: 'ok', message: 'API работает' }; } catch (err) { result = { status: 'error', message: 'Ошибка: ' + err.message }; } const body = cb ? cb + '(' + JSON.stringify(result) + ')' : JSON.stringify(result); const type = cb ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON; return ContentService.createTextOutput(body).setMimeType(type); } function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.appendRow(HEADERS); sh.setFrozenRows(1); } if (sh.getLastRow() === 0) { sh.appendRow(HEADERS); sh.setFrozenRows(1); } return sh; } function register(p) { const email = String(p.email || '').trim().toLowerCase(); const pass = String(p.pass || '').trim(); const name = String(p.name || '').trim(); if (!email || !pass || !name) return { status: 'error', message: 'Заполните все поля' }; if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) return { status: 'error', message: 'Неверный логин' }; if (pass.length < 6) return { status: 'error', message: 'Пароль минимум 6 символов' }; const sh = getSheet(); const last = sh.getLastRow(); if (last > 1) { const emails = sh.getRange(2, 1, last - 1, 1).getValues(); for (let i = 0; i < emails.length; i++) { if (String(emails[i][0]).trim().toLowerCase() === email) return { status: 'error', message: 'Такой пользователь уже зарегистрирован' }; } } sh.appendRow([email, pass, name, new Date().toISOString()]); return { status: 'ok', user: { email: email, name: name } }; } function login(p) { const email = String(p.email || '').trim().toLowerCase(); const pass = String(p.pass || '').trim(); if (!email || !pass) return { status: 'error', message: 'Заполните поля' }; const sh = getSheet(); const last = sh.getLastRow(); if (last < 2) return { status: 'error', message: 'Нет пользователей' }; const rows = sh.getRange(2, 1, last - 1, 3).getValues(); for (let i = 0; i < rows.length; i++) { if (String(rows[i][0]).trim().toLowerCase() === email && String(rows[i][1]).trim() === pass) { const n = String(rows[i][2] || '').trim() || email.split('@')[0]; return { status: 'ok', user: { email: email, name: n } }; } } return { status: 'error', message: 'Неверный логин или пароль' }; }
